import React from 'react'

interface WeatherData {
  temp: number
  feels_like: number
  description: string
  humidity: number
  wind_speed: number
  icon: string // simple emoji key
}

export default function WeatherCard({ data }: { data: WeatherData }) {
  return (
    <section className="card">
      <div className="card-left">
        <div className="temperature">{Math.round(data.temp)}°C</div>
        <div className="description">{data.description}</div>
        <div className="meta">
          <span>Feels {Math.round(data.feels_like)}°</span>
          <span>Humidity {data.humidity}%</span>
          <span>Wind {data.wind_speed} km/h</span>
        </div>
      </div>
      <div className="card-right">
        <div className="icon" aria-hidden>
          {renderIcon(data.icon)}
        </div>
      </div>
    </section>
  )
}

function renderIcon(name: string) {
  const icons: Record<string, string> = {
    sunny: '☀️',
    cloudy: '☁️',
    rain: '🌧️',
    snow: '❄️',
    storm: '⛈️',
  }
  return icons[name] ?? '🌤️'
}

interface ForecastItem {
  time: string
  temp: number
  icon: string
}

export default function ForecastList({ forecasts }: { forecasts: ForecastItem[] }) {
  const labels = forecasts.map((f) => f.time)
  const data = {
    labels,
    datasets: [
      {
        label: 'Temperature (°C)',
        data: forecasts.map((f) => f.temp),
        borderColor: '#3b82f6',
        backgroundColor: 'rgba(59,130,246,0.2)',
        tension: 0.4,
      },
    ],
  }

  return (
    <aside className="forecast">
      <h2>Hourly Forecast</h2>
      <div className="forecast-grid">
        {forecasts.map((f) => (
          <div key={f.time} className="forecast-item">
            <div className="time">{f.time}</div>
            <div className="icon">{renderIcon(f.icon)}</div>
            <div className="temp">{Math.round(f.temp)}°</div>
          </div>
        ))}
      </div>
      <div className="chart">
        <Line data={data} />
      </div>
    </aside>
  )
}

function renderIcon(name: string) {
  const icons: Record<string, string> = {
    sunny: '☀️',
    cloudy: '☁️',
    rain: '🌧️',
    snow: '❄️',
    storm: '⛈️',
  }
  return icons[name] ?? '🌤️'
}
